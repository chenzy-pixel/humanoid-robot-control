cmake_minimum_required(VERSION 3.16)
project(usb2can_demo_lingzu)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The USB2CAN C++ transport requires Linux.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(Threads REQUIRED)
add_executable(can_node src/Tangair_usb2can.cpp src/can_node.cpp ../../../common/src/usb_can.cpp)
target_include_directories(can_node PRIVATE include can ../../../common/include)
target_compile_options(can_node PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(can_node PRIVATE Threads::Threads)
ament_target_dependencies(can_node PUBLIC rclcpp rcl_interfaces sensor_msgs)
install(TARGETS can_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY ../../../launch ../../../config DESTINATION share/${PROJECT_NAME})
install(FILES can/usb_can.rules DESTINATION share/${PROJECT_NAME}/udev)

if(BUILD_TESTING)
    find_package(ament_cmake_gtest REQUIRED)
    ament_add_gtest(config_test test/config_test.cpp src/Tangair_usb2can.cpp ../../../common/src/usb_can.cpp)
    target_include_directories(config_test PRIVATE include can ../../../common/include)
    target_compile_definitions(config_test PRIVATE MOTOR_CONFIG_FILE="${CMAKE_CURRENT_SOURCE_DIR}/../../../config/motors.yaml")
    target_link_libraries(config_test Threads::Threads)
    ament_target_dependencies(config_test rclcpp rcl_interfaces sensor_msgs)
endif()
ament_package()
