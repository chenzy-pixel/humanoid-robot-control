cmake_minimum_required(VERSION 3.16)
project(usb2can_demo_lingzu)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The USB2CAN C++ transport requires Linux.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(control_msgs REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(Threads REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
    "srv/JointCommand.srv" "msg/TrajectoryTask.msg" DEPENDENCIES std_msgs)
rosidl_get_typesupport_target(controller_typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
include(../../../common/ruckig.cmake)
add_library(joint_motion STATIC ../../../common/src/joint_motion.cpp)
target_include_directories(joint_motion PUBLIC ../../../common/include)
target_link_libraries(joint_motion PUBLIC robot_ruckig)
set(CONTROLLER_SOURCES src/Tangair_usb2can.cpp src/controller_api.cpp src/controller_motion.cpp)
add_executable(can_node ${CONTROLLER_SOURCES} src/can_node.cpp ../../../common/src/usb_can.cpp)
target_include_directories(can_node PRIVATE include can ../../../common/include)
target_compile_options(can_node PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(can_node PRIVATE Threads::Threads joint_motion "${controller_typesupport}")
ament_target_dependencies(can_node PUBLIC rclcpp rclcpp_action control_msgs rcl_interfaces sensor_msgs trajectory_msgs std_msgs)
install(TARGETS can_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY ../../../launch ../../../config DESTINATION share/${PROJECT_NAME})
install(FILES can/usb_can.rules DESTINATION share/${PROJECT_NAME}/udev)
install(FILES ../../../third_party/ruckig/LICENSE DESTINATION share/${PROJECT_NAME}/licenses RENAME Ruckig-MIT.txt)

if(BUILD_TESTING)
    find_package(ament_cmake_gtest REQUIRED)
    ament_add_gtest(config_test test/config_test.cpp ${CONTROLLER_SOURCES} ../../../common/src/usb_can.cpp)
    target_include_directories(config_test PRIVATE include can ../../../common/include)
    target_compile_definitions(config_test PRIVATE MOTOR_CONFIG_FILE="${CMAKE_CURRENT_SOURCE_DIR}/../../../config/motors.yaml")
    target_link_libraries(config_test Threads::Threads joint_motion "${controller_typesupport}")
    ament_target_dependencies(config_test rclcpp rclcpp_action control_msgs rcl_interfaces sensor_msgs trajectory_msgs std_msgs)
    add_executable(trajectory_test ../../../tests/cpp/trajectory_test.cpp)
    target_link_libraries(trajectory_test PRIVATE joint_motion)
    target_compile_options(trajectory_test PRIVATE -UNDEBUG)
    add_test(NAME joint_trajectory_limits COMMAND trajectory_test)
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
